rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users collection - for user profiles
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      allow read: if request.auth != null; // Allow reading other users for username checks
    }
    
    // Notes collection - for user notes
    match /notes/{noteId} {
      allow read, update, delete: if 
        request.auth != null && 
        request.auth.uid == resource.data.userId;
        
      allow create: if 
        request.auth != null && 
        request.auth.uid == request.resource.data.userId;
        
      match /blocks/{blockId} {
        allow read, create, update, delete: if 
          request.auth != null &&
          get(/databases/$(database)/documents/notes/$(noteId)).data.userId == request.auth.uid;
      }
    }
  }
}
